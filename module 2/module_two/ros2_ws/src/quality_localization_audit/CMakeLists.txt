cmake_minimum_required(VERSION 3.8)
project(quality_localization_audit)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(tf_authority_audit src/tf_authority_audit.cpp)
target_compile_features(tf_authority_audit PUBLIC cxx_std_17)
target_compile_options(tf_authority_audit PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(tf_authority_audit rclcpp tf2_msgs std_msgs)
install(TARGETS tf_authority_audit DESTINATION lib/${PROJECT_NAME})
ament_package()
